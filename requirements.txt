# Planner Desktop (PySide6 + QML) — runtime dependencies.
# Install with:  python -m pip install -r requirements.txt

# UI: Qt Quick / QML application shell (planner_desktop/qml).
PySide6

# Google OAuth + Calendar API. Imported lazily inside
# planner_desktop/sync/google_auth.py and google_calendar_gateway.py:
# nothing here is touched until the user explicitly connects an account.
google-auth
google-auth-oauthlib
google-api-python-client

# Timezone database for the stdlib `zoneinfo` used by recurrence/series
# logic. Windows ships no system tz database, so this is required there.
tzdata; sys_platform == "win32"

# Optional: better local-timezone detection in
# planner_desktop/domain/recurrence.py. Imported inside a try/except —
# the app falls back to TZ / /etc/timezone / UTC when it is absent.
tzlocal

# Storage uses the standard-library `sqlite3` module: no ORM dependency.
